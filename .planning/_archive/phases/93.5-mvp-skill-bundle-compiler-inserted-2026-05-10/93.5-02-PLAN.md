    ```python
    """Compile-time skill-bundle composer.

    Reads detected intents (set[str] from _classify_user_intent per D-01
    supersession), composes a single narrator prompt + tool allowlist +
    citation allowlist by unioning the activated bundles' fragments per
    CONTEXT D-09 / D-10 / D-11 / D-12 / D-13 / D-14.

    Per CONTEXT D-04 — the static `_INTENT_BUNDLES` dict IS the resolver.
    Per RESEARCH §3.5 — single entrypoint `compile_bundles`.
    Per H4 fix (revision iter1 2026-05-10) — post-assembly undeclared-slot
    guard raises ValueError to prevent runtime KeyError in narrator call.
    """
    from __future__ import annotations

    import re
    from dataclasses import dataclass
    from typing import Iterable, Optional

    from app.services.coach.bundles import (
        BundleBase,
        ComplianceNarratorBundle,
        LifeEventRouterBundle,
        Pillar3aOptimizerBundle,
        LppProjectorBundle,
        TaxExplainerBundle,
        MortgageStressorBundle,
    )
    from app.services.coach.coach_models import CoachContext


    # Per CONTEXT D-02 — static intent → bundle mapping.
    _INTENT_BUNDLES: dict[str, list[type[BundleBase]]] = {
        "retirement": [Pillar3aOptimizerBundle, LppProjectorBundle],
        "taxes":      [TaxExplainerBundle, Pillar3aOptimizerBundle],
        "housing":    [MortgageStressorBundle, TaxExplainerBundle],
        "debt":       [MortgageStressorBundle, ComplianceNarratorBundle],
        "family":     [LifeEventRouterBundle, ComplianceNarratorBundle],
        "career":     [LppProjectorBundle, LifeEventRouterBundle],
    }

    # Per CONTEXT D-09 — always-on, never dropped.
    _ALWAYS_ON: list[type[BundleBase]] = [
        ComplianceNarratorBundle,
        LifeEventRouterBundle,
    ]

    # Per CONTEXT D-13 — drop order right-to-left when over budget.
    _DROP_PRIORITY: list[type[BundleBase]] = [
        MortgageStressorBundle,
        TaxExplainerBundle,
        LppProjectorBundle,
        Pillar3aOptimizerBundle,
    ]

    # Module-import-time defense-in-depth for T-93.5-06 (Pitfall 3).
    assert set(_DROP_PRIORITY).isdisjoint(set(_ALWAYS_ON)), (
        "_DROP_PRIORITY must be disjoint from _ALWAYS_ON"
    )

    # Per CONTEXT D-13 — final prompt size budget in tokens.
    _TOKEN_BUDGET: int = 8_000

    # Per CONTEXT D-11 — fragment join separator.
    _FRAGMENT_SEPARATOR: str = "\n\n---\n\n"

    # Allowed intents — same 6-enum as ExtractorOutput.intents.
    _ALLOWED_INTENTS: frozenset[str] = frozenset(_INTENT_BUNDLES.keys())

    # Per H4 fix — canonical 7-slot set provided by _build_prompt at
    # claude_coach_service.py:789. Any 8th slot in a bundle fragment
    # would raise KeyError at narrator call time.
    _DECLARED_SLOTS: frozenset[str] = frozenset({
        "banned_terms",
        "regional_identity",
        "lifecycle_awareness",
        "plan_awareness",
        "check_in_protocol",
        "safe_mode_protocol",
        "routing_rules",
    })

    # Single-brace `{slot}` pattern (ignores `{{cite:<key>}}` double-brace).
    _SINGLE_BRACE_SLOT = re.compile(r"(?<!\{)\{(\w+)\}(?!\})")


    @dataclass(frozen=True)
    class CompiledBundle:
        """Output of compile_bundles consumed by claude_coach_service."""
        prompt: str
        allowed_tools: list[str]
        citation_allowlist: list[str]
        activated_bundles: list[str]   # for telemetry (Sentry breadcrumb)
        estimated_tokens: int
        dropped_bundles: list[str]     # for diagnostics


    def _estimate_tokens(s: str) -> int:
        """4-char-per-token heuristic per RESEARCH §Pitfall 2."""
        return len(s) // 4


    def _assemble(bundles: list[BundleBase]) -> tuple[str, int]:
        fragments = [b.prompt_fragment for b in bundles]
        prompt_str = _FRAGMENT_SEPARATOR.join(fragments)
        return prompt_str, _estimate_tokens(prompt_str)


    def compile_bundles(
        intents: Iterable[str],
        ctx: Optional[CoachContext] = None,
        language: str = "fr",
    ) -> CompiledBundle:
        """Compose 6 bundles into a single narrator prompt.

        Per CONTEXT D-11/D-12/D-13/D-14. H4 — raises ValueError if assembled
        prompt contains an undeclared `{slot}` (defense-in-depth vs runtime
        KeyError in `_build_prompt.format(...)`).
        """
        # Step 1: defensively normalize intents (silently drop unknowns).
        normalized = set(i for i in intents if i in _ALLOWED_INTENTS)

        # Step 2: collect bundle classes (always-on first, intent-driven deduped).
        bundle_classes: list[type[BundleBase]] = list(_ALWAYS_ON)
        seen: set[type[BundleBase]] = set(_ALWAYS_ON)
        for intent in sorted(normalized):  # sorted for determinism
            for cls in _INTENT_BUNDLES.get(intent, []):
                if cls not in seen:
                    bundle_classes.append(cls)
                    seen.add(cls)

        # Step 3: instantiate.
        activated: list[BundleBase] = [cls() for cls in bundle_classes]

        # Step 4: assemble + drop right-to-left if over budget. Always-on
        # NEVER dropped (T-93.5-06 mitigation).
        prompt, est_tokens = _assemble(activated)
        dropped: list[str] = []
        while est_tokens > _TOKEN_BUDGET:
            droppable = [b for b in activated if type(b) in _DROP_PRIORITY]
            if not droppable:
                break  # only always-on left, accept overflow
            for drop_cls in _DROP_PRIORITY:
                target = next((b for b in activated if isinstance(b, drop_cls)), None)
                if target is not None:
                    activated.remove(target)
                    dropped.append(drop_cls.__name__)
                    break
            prompt, est_tokens = _assemble(activated)

        # Step 5 (H4 NEW): scan for undeclared slots before returning.
        slots_found = set(_SINGLE_BRACE_SLOT.findall(prompt))
        undeclared = slots_found - _DECLARED_SLOTS
        if undeclared:
            raise ValueError(
                f"compile_bundles: undeclared slot(s) {undeclared} found in "
                f"assembled prompt. Allowed: {_DECLARED_SLOTS}. "
                f"Activated bundles: {[b.name for b in activated]}."
            )

        # Step 6: dedup tools + citations (D-12), sorted for determinism.
        allowed_tools = sorted({t for b in activated for t in b.allowed_tools})
        citation_allowlist = sorted({c for b in activated for c in b.citation_allowlist})

        return CompiledBundle(
            prompt=prompt,
            allowed_tools=allowed_tools,
            citation_allowlist=citation_allowlist,
            activated_bundles=[b.name for b in activated],
            estimated_tokens=est_tokens,
            dropped_bundles=dropped,
        )


    __all__ = ["CompiledBundle", "compile_bundles"]
    ```
