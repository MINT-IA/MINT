    ```python
    _TOKEN_CACHE_PATH = (

    def _load_token_cache() -> dict[str, int]:
        if _TOKEN_CACHE_PATH.exists():
            return json.loads(_TOKEN_CACHE_PATH.read_text())
        return {}

    def _save_token_cache(cache: dict[str, int]) -> None:
        _TOKEN_CACHE_PATH.write_text(json.dumps(cache, indent=2, sort_keys=True))

    def count_tokens_cached(prompt: str, model: str, anthropic_client) -> int:
        key = hashlib.sha256(f"{model}::{prompt}".encode()).hexdigest()
        cache = _load_token_cache()
        if key in cache:
            return cache[key]
        try:
            result = anthropic_client.messages.count_tokens(
                model=model, system=prompt,
                messages=[{"role": "user", "content": "."}],
            )
            cache[key] = result.input_tokens
            _save_token_cache(cache)
            return result.input_tokens
        except Exception:  # noqa: BLE001 — fallback to heuristic
            return len(prompt) // 4
    ```
