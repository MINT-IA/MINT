    ```python
    #!/usr/bin/env python3
    """Banned-terms LSFin lint for Python source files (Phase 93.5).

    Scans `services/backend/app/services/coach/bundles/*.py` for banned
    LSFin narrator-output terms. Exemption : a multi-line string literal
    is exempted from the scan if the line IMMEDIATELY ABOVE it contains
    the marker `# llm-doctrine-fragment-banned-list`. The exemption only
    applies to the next assignment (e.g. `_PROMPT_FRAGMENT = \"\"\"...\"\"\"`).

    Rationale : the compliance-narrator bundle's prompt_fragment must
    LIST the banned terms verbatim to forbid them to the narrator LLM.
    Without exemption, the lint would falsely flag the doctrine block.
    Per CONTEXT D-09 + RESEARCH §Pitfall 4.

    Exit codes :
      0 — no banned terms outside exempted blocks
      1 — at least one banned term in non-exempted code
    """
    import argparse
    import re
    import sys
    from pathlib import Path

    BANNED = ["garanti", "optimal", "meilleur", "certain", "assure", "sans risque", "parfait"]
    EXEMPTION_MARKER = "# llm-doctrine-fragment-banned-list"

    def scan_file(path: Path) -> list[tuple[int, str, str]]:
        """Return list of (line_no, term, line_text) for non-exempted hits."""
        text = path.read_text(encoding="utf-8")
        lines = text.splitlines()
        hits: list[tuple[int, str, str]] = []
        # Compute exempted line ranges : from the line AFTER the marker through
        # the closing triple-quote of the next multi-line string (or next blank line).
        exempted_lines: set[int] = set()
        i = 0
        while i < len(lines):
            if EXEMPTION_MARKER in lines[i]:
                # Look ahead for next triple-quoted string opening
                j = i + 1
                in_block = False
                while j < len(lines):
                    if not in_block and ('"""' in lines[j] or "'''" in lines[j]):
                        in_block = True
                        exempted_lines.add(j)
                        # Single-line triple-quoted ?
                        if lines[j].count('"""') >= 2 or lines[j].count("'''") >= 2:
                            break
                    elif in_block:
                        exempted_lines.add(j)
                        if '"""' in lines[j] or "'''" in lines[j]:
                            break
                    j += 1
                i = j + 1
            else:
                i += 1
        for line_no, line in enumerate(lines, start=1):
            if line_no in exempted_lines:
                continue
            lower = line.lower()
            for term in BANNED:
                # Word-boundary scan to avoid matching inside e.g. "incertain"
                if re.search(r"\b" + re.escape(term) + r"\b", lower):
                    hits.append((line_no, term, line.rstrip()))
                    break
        return hits

    def main(argv: list[str] | None = None) -> int:
        parser = argparse.ArgumentParser(description="Banned-terms Python lint")
        parser.add_argument("paths", nargs="+", type=Path)
        args = parser.parse_args(argv)
        all_hits: list[tuple[Path, int, str, str]] = []
        for p in args.paths:
            if p.is_dir():
                for f in p.rglob("*.py"):
                    for line_no, term, line in scan_file(f):
                        all_hits.append((f, line_no, term, line))
            elif p.is_file():
                for line_no, term, line in scan_file(p):
                    all_hits.append((p, line_no, term, line))
        if all_hits:
            for path, line_no, term, line in all_hits:
                print(f"{path}:{line_no}: banned term {term!r}: {line}", file=sys.stderr)
            return 1
        return 0

    if __name__ == "__main__":
        sys.exit(main())
    ```
