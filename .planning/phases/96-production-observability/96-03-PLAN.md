`services/backend/app/utils/sentry_audit_tags.py` (~50 lines):

```python
"""Phase 96-03 OBS-03 — Sentry audit tag helper.

Single source of truth for the 4-tag pattern emitted on every coach
response (authed /coach/chat + anonymous /anonymous/chat + FATCA
handoff path). Best-effort: any exception is swallowed (the user
response must NEVER break because Sentry is misconfigured).
"""
from __future__ import annotations
from typing import Optional

def _emit_sentry_audit_tags(
    *,
    eclairage_kind: Optional[str],
    banned_term_hit: bool,
    eval_score: float,
    archetype: Optional[str],
) -> None:
    """Set 4 Sentry tags on the in-flight transaction. Best-effort."""
    try:
        import sentry_sdk
        try:
            sentry_sdk.set_tag("eclairage_kind", (eclairage_kind or "none"))
            sentry_sdk.set_tag("banned_term_hit", "true" if banned_term_hit else "false")
            sentry_sdk.set_tag("eval_score", f"{eval_score:.2f}")
            sentry_sdk.set_tag("archetype", (archetype or "unknown"))
        except Exception:
            pass
    except ImportError:
        pass
```
