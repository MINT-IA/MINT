    def check_constants_drift(dart_file: Path, *, hard: bool, json_status: bool) -> int:
        """Compare baked Dart version_hash against backend RegulatoryRegistry.version_hash(today).

        Returns exit code: 0 (OK or MISSING-GENERATED-FILE or SOFT drift), 1 (HARD drift), 2 (import failure).
        """
        import json as _json

        # Missing-file path — graceful (Plan 04 not merged yet).
        if not dart_file.exists():
            _emit_status("constants-drift", "MISSING-GENERATED-FILE", 0,
                         f"Generated Dart file not found at {dart_file} — Plan 04 not yet merged ; constants-drift check skipped.",
                         json_status)
            return 0

        # Import path — LOUD failure on env issue (codex MEDIUM #2).
        try:
            backend_root = _REPO_ROOT / "services" / "backend"
            if str(backend_root) not in sys.path:
                sys.path.insert(0, str(backend_root))
            from app.services.regulatory.registry import RegulatoryRegistry  # noqa: WPS433
        except ImportError as exc:
            _emit_status("constants-drift", "IMPORT-FAILURE", 2,
                         f"Failed to import RegulatoryRegistry from services/backend ({exc}). Verify backend deps are installed.",
                         json_status)
            return 2

        # Extract baked hash from Dart file.
        dart_text = dart_file.read_text(encoding="utf-8")
        match = re.search(r"const String regulatoryConstantsVersionHash = '([0-9a-f]{64})';", dart_text)
        if not match:
            _emit_status("constants-drift", "IMPORT-FAILURE", 2,
                         f"Could not parse regulatoryConstantsVersionHash from {dart_file}. Re-run Plan 04 codegen.",
                         json_status)
            return 2
        baked = match.group(1)

        # Compare to backend.
        from datetime import date
        backend_hash = RegulatoryRegistry.instance().version_hash(date.today())
        if baked == backend_hash:
            _emit_status("constants-drift", "OK", 0, None, json_status)
            return 0

        # Drift detected.
        exit_code = 1 if hard else 0
        suffix = " (HARD)" if hard else " (SOFT-WARN — Phase 01 D-12 ; Phase 02 will promote to HARD)"
        _emit_status(
            "constants-drift", "DRIFT-CONSTANTS", exit_code,
            f"Drift detected{suffix}: Dart baked = {baked} ; backend version_hash = {backend_hash}. "
            f"Fix: python3 tools/codegen/regulatory_constants_to_dart.py --source local --write",
            json_status,
        )
        return exit_code


    def _emit_status(mode: str, status: str, exit_code: int, msg: str | None, json_status: bool) -> None:
        if msg:
            print(f"[{mode}] {status}: {msg}", file=sys.stderr)
        if json_status:
            import json as _json
            print(_json.dumps({"mode": mode, "status": status, "exit_code": exit_code}), file=sys.stderr)
