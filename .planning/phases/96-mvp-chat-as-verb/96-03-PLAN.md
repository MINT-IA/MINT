       ```python
       #!/usr/bin/env python3
       """Phase 96 D-17 — metaphor TOML byte-equality + value-poisoning lint.

       Compares apps/mobile/assets/metaphors.toml and
       services/backend/app/data/metaphors.toml — must be byte-equal (sha256).

       With --scan-values, also scans every metaphor string value for :
       - LSFin banned terms (« garanti », « optimal », « parfait », ...) via banned_terms_python rules
       - PII patterns (email regex, phone regex, IBAN regex)

       Exit codes :
         0 — clean
         1 — parity violation OR poisoned value
       """
       import hashlib
       import re
       import sys
       import tomllib
       from pathlib import Path

       MOBILE = Path("apps/mobile/assets/metaphors.toml")
       BACKEND = Path("services/backend/app/data/metaphors.toml")

       _BANNED_LSFIN = [
           "garanti", "garantie", "garantis",
           "optimal", "optimale", "optimaux",
           "parfait", "parfaite",
           "meilleur", "meilleure",
           "certain", "certaine",
           "assuré", "assurée",
           "sans risque",
       ]
       _PII_REGEXES = [
           re.compile(r"[\w.+-]+@[\w-]+\.[\w.-]+"),  # email
           re.compile(r"\+?\d{10,15}"),               # phone
           re.compile(r"CH\d{2}\s?\d{4}\s?\d{4}"),    # IBAN prefix
       ]


       def sha256(path: Path) -> str:
           return hashlib.sha256(path.read_bytes()).hexdigest()


       def scan_values(path: Path) -> list[str]:
           violations = []
           data = tomllib.loads(path.read_text())
           for archetype, cantons in data.items():
               if not isinstance(cantons, dict):
                   continue
               for canton, events in cantons.items():
                   if not isinstance(events, dict):
                       continue
                   for event, payload in events.items():
                       if not isinstance(payload, dict):
                           continue
                       metaphor = payload.get("metaphor", "")
                       lower = metaphor.lower()
                       for banned in _BANNED_LSFIN:
                           if banned in lower:
                               violations.append(
                                   f"{archetype}.{canton}.{event}: "
                                   f"LSFin banned term « {banned} » in metaphor"
                               )
                       for regex in _PII_REGEXES:
                           if regex.search(metaphor):
                               violations.append(
                                   f"{archetype}.{canton}.{event}: "
                                   f"PII pattern detected"
                               )
           return violations


       def main(argv: list[str]) -> int:
           if sha256(MOBILE) != sha256(BACKEND):
               print("ERROR: mobile + backend metaphors.toml diverged", file=sys.stderr)
               print(f"  mobile : {sha256(MOBILE)}")
               print(f"  backend: {sha256(BACKEND)}")
               return 1
           if "--scan-values" in argv:
               violations = scan_values(MOBILE)
               if violations:
                   for v in violations:
                       print(f"ERROR: {v}", file=sys.stderr)
                   return 1
           print("OK")
           return 0


       if __name__ == "__main__":
           sys.exit(main(sys.argv[1:]))
       ```

       ```python
       Runs AFTER the citation gate (Phase 94 stays first in middleware chain).
       If hook contains any digit (regex \\d), swap to the generic fallback.
       NEVER raise. NEVER 500.

       T-96-W3-LinterDoS mitigation :
       - Regex is r"\\d" — a simple character class, no catastrophic backtracking.
       - signal.alarm(0.1) wraps the .search() call for defense-in-depth.
       - On any exception (timeout, regex error, sentry error), fall back to
         the generic hook — fail-safe, never fail-open.
       """

       import re
       import signal
       import sentry_sdk
       from app.schemas.narrative_sleeve import NarrativeSleeve

       _DIGIT_RE = re.compile(r"\d", re.UNICODE)

       # D-16 verbatim FR — accent_lint clean, LSFin clean.
       HOOK_FALLBACK: str = "Voyons ensemble ce que ça change pour toi."


       class _LintTimeout(Exception):
           pass


       def _timeout_handler(signum, frame):  # noqa: ARG001
           raise _LintTimeout()


       def _has_digit_with_timeout(hook: str, timeout_ms: int = 100) -> bool:
           """Run the regex search with a SIGALRM-based timeout.

           Returns True on digit match. Returns True (force fallback) on timeout —
           if the regex hangs on adversarial input, the safe choice is to swap.
           """
           # signal.SIGALRM is POSIX-only — guard for Windows test runners.
           if not hasattr(signal, "SIGALRM"):
               return bool(_DIGIT_RE.search(hook))
           old_handler = signal.signal(signal.SIGALRM, _timeout_handler)
           signal.setitimer(signal.ITIMER_REAL, timeout_ms / 1000.0)
           try:
               return bool(_DIGIT_RE.search(hook))
           except _LintTimeout:
               return True  # Defensive : on timeout, force fallback.
           finally:
               signal.setitimer(signal.ITIMER_REAL, 0)
               signal.signal(signal.SIGALRM, old_handler)


       def lint_sleeve(sleeve: NarrativeSleeve) -> NarrativeSleeve:
           """Swap hook to fallback if it contains digits.

           NarrativeSleeve is frozen — model_copy(update=...) is the only
           mutation path. Returns the SAME instance on clean input.
           """
           try:
               needs_swap = _has_digit_with_timeout(sleeve.hook)
           except Exception:  # noqa: BLE001 — defense-in-depth
               needs_swap = True  # Defensive : on any error, force fallback.

           if not needs_swap:
               return sleeve

                   level="info",
                   data={"original_hook_length": len(sleeve.hook)},
               )
           except Exception:  # pragma: no cover — breadcrumb is fail-open
               pass
           return sleeve.model_copy(update={"hook": HOOK_FALLBACK})
       ```
