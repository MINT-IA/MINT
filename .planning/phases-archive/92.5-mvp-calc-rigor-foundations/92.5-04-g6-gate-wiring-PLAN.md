#!/usr/bin/env python3
"""G6 calc-correctness path decider — CALC-04 (CONTEXT 92.5 D-18/D-21).

Deterministic CLI: given a list of changed file paths (from --files
or --diff-base ref), exit 0 if ANY path matches a G6 trigger path,
exit 1 otherwise. Used by:
  (a) .github/workflows/calc-rigor.yml as belt-and-suspenders alongside
      the GitHub Actions `paths:` filter,
  (b) GSD verifier integration (D-21): any phase whose plan touches
      the trigger paths must invoke this tool to confirm G6 applies.

Trigger paths (D-18 EXACT — no expansion without an ADR):
  - apps/mobile/lib/services/financial_core/**
  - services/backend/app/services/**
  - services/backend/app/constants/social_insurance.py

Mirrors `tools/checks/accent_lint_fr.py` in style: argparse + clean exit
codes + machine-readable stderr.
"""
from __future__ import annotations

import argparse
import fnmatch
import subprocess
import sys
from pathlib import Path

# D-18 EXACT trigger paths. To extend, write an ADR + bump the doc.
TRIGGER_GLOBS: list[str] = [
    "apps/mobile/lib/services/financial_core/*",
    "apps/mobile/lib/services/financial_core/**/*",
    "services/backend/app/services/*",
    "services/backend/app/services/**/*",
    "services/backend/app/constants/social_insurance.py",
]


def matches_trigger(path: str) -> bool:
    p = path.lstrip("./")
    for glob in TRIGGER_GLOBS:
        if fnmatch.fnmatch(p, glob):
            return True
    return False


def files_from_diff(base: str) -> list[str]:
    """Return changed files between HEAD and base ref via git."""
    out = subprocess.run(
        ["git", "diff", "--name-only", f"{base}...HEAD"],
        capture_output=True,
        text=True,
        check=True,
    )
    return [line.strip() for line in out.stdout.splitlines() if line.strip()]


def main() -> int:
    parser = argparse.ArgumentParser(description="G6 calc-correctness path decider (CALC-04).")
    src = parser.add_mutually_exclusive_group(required=True)
    src.add_argument("--files", nargs="+", help="Explicit list of changed file paths.")
    src.add_argument("--diff-base", help="Git ref (e.g. origin/dev) ; uses git diff --name-only.")
    parser.add_argument(
        "--quiet",
        action="store_true",
        help="Suppress stderr summary lines.",
    )
    args = parser.parse_args()

    if args.files:
        files = list(args.files)
    else:
        files = files_from_diff(args.diff_base)

    matched = [f for f in files if matches_trigger(f)]

    if not args.quiet:
        if matched:
            print(f"[g6] applies — {len(matched)} trigger-path file(s):", file=sys.stderr)
            for f in matched:
                print(f"[g6]   {f}", file=sys.stderr)
        else:
            print(
                f"[g6] does not apply — none of {len(files)} changed file(s) match D-18 trigger paths.",
                file=sys.stderr,
            )

    return 0 if matched else 1


if __name__ == "__main__":
    sys.exit(main())
```

# MINT calc-rigor CI gate — Phase 92.5 / CALC-04 (CONTEXT D-07/D-18/D-19/D-20).
#
# Triggers ONLY on PRs / pushes that touch one of the 3 D-18 trigger paths
# (financial_core, backend services, social_insurance.py). Other PRs skip
# this workflow entirely — G1-G5 remain the universal gates.
#
# Hard-block from day 1 (D-19): no warn-only ramp-up, because silent drift
# is the failure mode this gate exists to prevent.

name: G6 calc-rigor

on:
  pull_request:
    branches: [dev, staging, main]
    paths:
      - 'apps/mobile/lib/services/financial_core/**'
      - 'services/backend/app/services/**'
      - 'services/backend/app/constants/social_insurance.py'
      # Self-edits to the harness / property suite / oracle / failure-comment also re-trigger
      - 'apps/mobile/tools/calc_harness/**'
      - 'services/backend/tests/test_calc_diff_harness.py'
      - 'services/backend/tests/test_property_invariants.py'
      - 'services/backend/tests/test_estv_oracle.py'
      - 'services/backend/tests/fixtures/calc_diff_v1.jsonl'
      - 'services/backend/tests/fixtures/estv_oracle_2025.jsonl'
      - 'tools/checks/estv_oracle_freshness.py'
      - 'tools/checks/g6_path_check.py'
      - '.github/workflows/calc-rigor.yml'
  push:
    branches: [dev, staging, main]
    paths:
      - 'apps/mobile/lib/services/financial_core/**'
      - 'services/backend/app/services/**'
      - 'services/backend/app/constants/social_insurance.py'

concurrency:
  group: calc-rigor-${{ github.ref }}
  cancel-in-progress: true

jobs:
  calc-rigor:
    name: G6 calc-correctness
    runs-on: ubuntu-latest
    permissions:
      contents: read
      pull-requests: write     # required to gh pr comment on failure
    steps:
      - uses: actions/checkout@v4

      # ─── Belt-and-suspenders: confirm the path filter would have triggered ──
      - name: Confirm G6 applies (g6_path_check)
        env:
          BASE: ${{ github.event.pull_request.base.ref || 'origin/dev' }}
        run: |
          set +e
          # On push events there's no PR base — use the trigger paths filter only.
          if [ "${{ github.event_name }}" = "pull_request" ]; then
            git fetch origin "${BASE}" --depth=1 || true
            python3 tools/checks/g6_path_check.py --diff-base "origin/${BASE}"
            rc=$?
          else
            echo "[g6] push event — path filter already gated this run."
            rc=0
          fi
          if [ "$rc" -ne 0 ]; then
            echo "::warning::g6_path_check disagrees with workflow trigger — investigate."
          fi
          # Don't fail on disagreement: the workflow trigger filter is the source of truth.
          exit 0

      # ─── Dart compile (cached) ────────────────────────────────────
      - uses: dart-lang/setup-dart@v1
        with:
          sdk: 3.5.4

      - name: Dart pub get
        working-directory: apps/mobile
        run: dart pub get

      - name: Compute financial_core checksum (for cache key)
        id: fc-hash
        run: |
          HASH=$(find apps/mobile/lib/services/financial_core apps/mobile/tools/calc_harness -type f \
            \( -name '*.dart' -o -name '*.yaml' \) -print0 \
            | sort -z | xargs -0 sha256sum | sha256sum | awk '{print $1}')
          echo "hash=${HASH}" >> "$GITHUB_OUTPUT"

      - name: Cache compiled calc_harness binary
        id: cache-harness
        uses: actions/cache@v4
        with:
          path: build/calc_harness_dart
          key: calc-harness-${{ runner.os }}-dart-3.5.4-${{ steps.fc-hash.outputs.hash }}

      - name: Compile calc_harness (Dart)
        if: steps.cache-harness.outputs.cache-hit != 'true'
        working-directory: apps/mobile
        run: |
          mkdir -p ../../build
          dart compile exe tools/calc_harness/main.dart -o ../../build/calc_harness_dart

      # ─── Python tests ─────────────────────────────────────────────
      - uses: actions/setup-python@v5
        with:
          python-version: '3.11'
          cache: pip
          cache-dependency-path: services/backend/pyproject.toml

      - name: Install backend deps
        working-directory: services/backend
        run: pip install ".[dev]"

      - name: Run differential harness
        env:
          CALC_HARNESS_BIN: ${{ github.workspace }}/build/calc_harness_dart
          TESTING: '1'
        working-directory: services/backend
        run: python3 -m pytest tests/test_calc_diff_harness.py -v --tb=short

      - name: Run property invariants
        env:
          TESTING: '1'
        working-directory: services/backend
        run: python3 -m pytest tests/test_property_invariants.py -v --tb=short

      - name: Run ESTV oracle (auto-skips if fixture empty)
        env:
          TESTING: '1'
        working-directory: services/backend
        run: python3 -m pytest tests/test_estv_oracle.py -v --tb=short

      - name: ESTV oracle freshness lint (WARN-only)
        run: python3 tools/checks/estv_oracle_freshness.py

      # ─── On failure: post failure-comment template to PR (D-20) ──
      - name: Post failure comment to PR
        if: failure() && github.event_name == 'pull_request'
        env:
          GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}
          PR_NUMBER: ${{ github.event.pull_request.number }}
        run: |
          gh pr comment "$PR_NUMBER" --body-file .github/workflows/calc-rigor-failure-comment.md \
            || echo "::warning::failed to post failure comment (continuing, do not gate on comment)"
```
